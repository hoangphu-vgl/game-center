function generateArenaStats(weight: number) {
  const rand = Math.random() * 100;
  let rank = 'D'; let statMult = 1.0;
  if (rand < 5) { rank = 'S'; statMult = 4.0; }
  else if (rand < 15) { rank = 'A'; statMult = 2.5; }
  else if (rand < 30) { rank = 'B'; statMult = 1.8; }
  else if (rand < 60) { rank = 'C'; statMult = 1.2; }
  else { rank = 'D'; statMult = 0.8; }

  const baseVal = Math.max(10, Math.sqrt(weight) * 15);
  return {
    rank,
    power: Math.floor(baseVal * statMult * (0.8 + Math.random() * 0.4)),
    defense: Math.floor(baseVal * statMult * (0.8 + Math.random() * 0.4)),
    arenaHp: Math.floor(baseVal * statMult * (1.5 + Math.random() * 0.5) * 5)
  };
}

      } else if (bs.state === "CLASH") {
        if (bs.timer > 30) {
          const isCrit = Math.random() < 0.15;
          const isEvade = Math.random() < 0.15;

          if (isEvade) {
            bs.p2X += 40;
            bs.floatTexts.push({ text: "NÉ ĐÒN!", x: bs.p2X, y: height * 0.65, color: "#38BDF8", life: 40 });
          } else {
            const p1Dmg = Math.max(15, (selectedFighter.power || 50) - (enemyFighter.defense || 30));
            const p2Dmg = Math.max(15, (enemyFighter.power || 50) - (selectedFighter.defense || 30));

            if (Math.random() > 0.5) {
              const finalDmg = Math.round(p1Dmg * (isCrit ? 2.4 : 1.2));
              bs.p2Hp = Math.max(0, bs.p2Hp - finalDmg);
              bs.floatTexts.push({ text: `${isCrit ? 'CRIT! ' : ''}-${finalDmg} HP`, x: bs.p2X, y: height * 0.65, color: isCrit ? "#FACC15" : "#EF4444", life: 50 });
              if (isCrit) bs.shakeTimer = 45;
            } else {
              const finalDmg = Math.round(p2Dmg * (isCrit ? 2.4 : 1.2));
              bs.p1Hp = Math.max(0, bs.p1Hp - finalDmg);
              bs.floatTexts.push({ text: `${isCrit ? 'CRIT! ' : ''}-${finalDmg} HP`, x: bs.p1X, y: height * 0.65, color: isCrit ? "#FACC15" : "#EF4444", life: 50 });
              if (isCrit) bs.shakeTimer = 45;
            }
          }

          setHpForceTick(prev => prev + 1);

          bs.p1X = bs.p1BaseX;
          bs.p2X = bs.p2BaseX;
          bs.p1Y = 0; bs.p2Y = 0;
          bs.timer = 0;
          bs.state = "CIRCLING";

          if (bs.p1Hp <= 0 || bs.p2Hp <= 0) {
            triggerWin(bs.p1Hp <= 0 ? enemyFighter.vnName : selectedFighter.vnName);
          }
        }
      } else if (bs.state === "ULT") {
        bs.activeUltTimer += speedMultiplier;
        if (bs.activeUltTimer > 60) {
          const isCrit = Math.random() < 0.15;
          if (bs.ultCaster === 1) {
            const dmg = Math.round((selectedFighter.power || 50) * (isCrit ? 3.75 : 2.5));
            bs.p2Hp = Math.max(0, bs.p2Hp - dmg);
            bs.floatTexts.push({ text: `${isCrit ? 'ULTI CRIT! ' : ''}-${dmg} HP`, x: bs.p2X, y: height * 0.65, color: "#FACC15", life: 60 });
          } else {
            const dmg = Math.round((enemyFighter.power || 50) * (isCrit ? 3.75 : 2.5));
            bs.p1Hp = Math.max(0, bs.p1Hp - dmg);
            bs.floatTexts.push({ text: `${isCrit ? 'ULTI CRIT! ' : ''}-${dmg} HP`, x: bs.p1X, y: height * 0.65, color: "#FACC15", life: 60 });
          }

          setHpForceTick(prev => prev + 1);

          bs.activeUltText = "";
          bs.timer = 0;
          bs.state = "CIRCLING";
          bs.shakeTimer = 45;

          if (bs.p1Hp <= 0 || bs.p2Hp <= 0) {
            triggerWin(bs.p1Hp <= 0 ? enemyFighter.vnName : selectedFighter.vnName);
          }
        }
      }

      if (bs.p1Hp <= 0 || bs.p2Hp <= 0) {
        triggerWin(bs.p1Hp <= 0 ? enemyFighter.vnName : selectedFighter.vnName);
      }
    };

      const bs = battleStateRef.current;

      });
      setGameState("result");

        setScore(newScore);
        supabase.from("profiles").update({ balance: newScore }).eq("id", user?.id);

      ctx.translate(bs.p1X, floorY - 50 + bs.p1Y);

      aura1.addColorStop(0.5, "rgba(202, 138, 4, 0.5)");
      aura1.addColorStop(1, "transparent");

      ctx.beginPath(); ctx.arc(0, 0, 105 + Math.sin(Date.now() * 0.015) * 15, 0, Math.PI * 2); ctx.fill();

      aura2.addColorStop(0.5, "rgba(168, 85, 247, 0.5)");

        beam.addColorStop(0.3, "#38BDF8");

        ctx.font = "900 24px sans-serif";
        ctx.fillStyle = "#FACC15";
        ctx.textAlign = "center";
        ctx.strokeStyle = "#000";
        ctx.lineWidth = 5;

                      className={`bg-slate-950 border-2 rounded-2xl p-3 flex flex-col items-center cursor-pointer transition ${

                    >

                selectedFighter && fishInventory.length > 0 ? "bg-gradient-to-r from-cyan-500 to-blue-600 text-slate-950 hover:brightness-110" : "bg-slate-800 text-slate-500 cursor-not-allowed"
              }`}

              🚀 BẮT ĐẦU VÕ ĐÀI (PVE)

      {/* SHOP SKILL */}

          <div className="max-w-3xl w-full bg-slate-900 border-2 border-purple-500/60 p-6 rounded-3xl shadow-2xl flex flex-col max-h-[85vh]">
            <div className="flex justify-between items-center pb-3 border-b border-slate-800 mb-3">
              <h3 className="text-lg font-black text-purple-400 uppercase">🛒 Shop Kỹ Năng & Mua Số Lượng Lớn</h3>
              <button onClick={() => setGameState("lobby")} className="bg-slate-800 text-white w-7 h-7 rounded-full font-black text-xs">✕</button>
            </div>
            <div className="flex-1 overflow-y-auto space-y-3 pr-1">
              {ALL_SKILLS.map((skill) => {
                const ownedQty = ownedSkillsCount[skill.id] || 0;
                const currentBuyQty = buyQuantities[skill.id] || 1;

                return (
                  <div key={skill.id} className="bg-slate-950 border border-slate-800 rounded-2xl p-3 flex flex-col sm:flex-row items-center justify-between gap-3">
                    <div className="flex items-center gap-3 w-full sm:w-auto">
                      <div className="text-3xl bg-slate-900 p-2 rounded-xl">{skill.icon}</div>

      {gameState === "skillInventory" && (

        <>
          <div className="absolute top-16 left-4 right-4 flex justify-between items-center z-30 pointer-events-none">
            <div className="bg-slate-900/90 border border-sky-500 p-2.5 rounded-2xl w-40 sm:w-56 shadow-lg">
              <div className="flex justify-between text-[10px] font-black mb-1">
                <span className="text-sky-300 truncate">{selectedFighter?.vnName}</span>
                <span className="text-rose-400">{Math.round(battleStateRef.current.p1Hp)}/{battleStateRef.current.p1MaxHp} HP</span>
              </div>

                <div className="bg-emerald-500 h-full transition-all" style={{ width: `${Math.max(0, Math.min(100, (battleStateRef.current.p1Hp / battleStateRef.current.p1MaxHp) * 100))}%` }}></div>

                <span>Nộ Ki</span>
                <span className="text-amber-300">{Math.round(battleStateRef.current.p1Ki)}%</span>

                <div className="bg-amber-400 h-full transition-all" style={{ width: `${battleStateRef.current.p1Ki}%` }}></div>

              <div className="flex justify-between text-[10px] font-black mb-1">
                <span className="text-rose-300 truncate">{enemyFighter?.vnName}</span>
                <span className="text-rose-400">{Math.round(battleStateRef.current.p2Hp)}/{battleStateRef.current.p2MaxHp} HP</span>

                const ownedQty = ownedSkillsCount[sId] || 0;

                return (
                  <button
                    key={sId}
                    disabled={cd > 0 || isUsed || (sObj.price > 0 && ownedQty <= 0)}
                    onClick={() => triggerSkill(sId)}
                    className={`px-3 py-1.5 rounded-xl flex flex-col items-center transition shadow relative ${
                      cd > 0 || isUsed || (sObj.price > 0 && ownedQty <= 0)
                        ? "bg-slate-900 border border-slate-800 opacity-50 cursor-not-allowed"
                        : "bg-slate-950 hover:bg-slate-800 border border-cyan-500/50 cursor-pointer"
                    }`}
                  >
                    <span className="text-base">{sObj.icon}</span>
                    <span className="text-[9px] font-black text-cyan-300">{sObj.name}</span>
                    <span className="text-[8px] text-amber-400 font-bold">
                      {isUsed ? "Đã dùng (1/1)" : cd > 0 ? `CD: ${cd}s` : sObj.price > 0 ? `Kho: ${ownedQty}` : "Sẵn sàng"}
                    </span>
                  </button>
                );
              })}
            </div>

            <div className="flex gap-1.5 bg-black/70 p-1.5 rounded-2xl border border-white/20 backdrop-blur">
              {[1, 2, 3].map((spd) => (
                <button
                  key={spd}
                  onClick={() => setSpeedMultiplier(spd)}
                  className={`px-2.5 py-1 rounded-xl text-[10px] font-black transition cursor-pointer ${
                    speedMultiplier === spd ? "bg-cyan-500 text-slate-950 shadow" : "bg-slate-800 text-slate-300"
                  }`}
                >
                  {spd}x
                </button>
              ))}
            </div>
          </div>
        </>
      )}

      {/* KẾT QUẢ TRẬN ĐẤU */}
      {gameState === "result" && matchResult && (
        <div className="absolute inset-0 z-50 bg-slate-950/90 backdrop-blur-md flex flex-col items-center justify-center p-4">
          <div className="max-w-md w-full bg-slate-900 border-2 border-amber-500 p-6 rounded-3xl shadow-2xl text-center space-y-4 animate-fadeIn">
            <h2 className={`text-2xl font-black uppercase ${matchResult.isWin ? "text-emerald-400" : "text-rose-500"}`}>{matchResult.message}</h2>
            <p className="text-xs text-slate-300">
              {matchResult.isWin ? `Nhận thưởng chiến thắng: +${matchResult.reward.toLocaleString()}🪙` : "Rèn luyện thêm sức mạnh ở Hồ Cá rồi quay lại phục thù!"}
            </p>
            <div className="flex gap-2 pt-2">
              <button
                onClick={() => setGameState("lobby")}
                className="flex-1 bg-cyan-500 hover:bg-cyan-400 text-slate-950 font-black py-3 rounded-xl text-xs uppercase shadow cursor-pointer transition"
              >
                🔄 Đấu Trận Khác
              </button>
              <Link
                href="/caucamulti/hoca"
                className="flex-1 bg-slate-800 hover:bg-slate-700 text-white font-black py-3 rounded-xl text-xs uppercase shadow text-center flex items-center justify-center transition"
              >
                🌊 Về Hồ Cá
              </Link>
            </div>
          </div>
        </div>
      )}

      <canvas ref={canvasRef} className="absolute inset-0 block w-full h-full -z-10 pointer-events-none"></canvas>
    </div>
  );
}
